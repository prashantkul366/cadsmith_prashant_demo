# Dependencies for the CADSmith web application.
#
# Installs the research pipeline's own requirements first, then what the app
# adds on top.  The pipeline's requirements.txt is left untouched.

-r ../requirements.txt

# Imported by autofab/metrics.py (KDTree) but missing from requirements.txt,
# so a fresh install from that file alone cannot compute CD/F1/IoU.
scipy

# Three-view rendering for the vision Judge.  autofab/render.py imports vtk
# directly; requirements.txt does not list it.
#
# On headless Linux the wheel imports fine and then fails at render time with
# no GL backend.  Install a software rasteriser as well:
#   Debian/Ubuntu:  sudo apt-get install libosmesa6
# macOS and Windows need nothing extra.  The app's health endpoint reports
# whether offscreen rendering actually works.
vtk

# Web server
fastapi
uvicorn[standard]

# Claude on Amazon Bedrock. The Anthropic SDK reaches Bedrock through boto3,
# and `pip install anthropic` alone does not bring it: without this the app
# still starts, the health banner still says Bedrock is ready, and the first
# Generate answers 503 blaming the AWS credentials - for a missing import.
anthropic[bedrock]

# Talks to OpenAI-compatible backends (OpenAI, Ollama, llama.cpp, vLLM,
# LM Studio, and the hosted gateways that speak the same API). The Anthropic
# path uses its own SDK, from requirements.txt.
httpx

# Makes Python trust the OS certificate store rather than only certifi.
# Required on corporate networks that inspect TLS, where every HTTPS
# connection is re-signed by a company CA the OS trusts and certifi does not.
truststore

# Writes the engineering drawing as DXF, which is the format a drawing is
# exchanged in. Its DIMENSION entities carry the geometry they measure, so a
# CAD system opening the file re-measures the part rather than reading back a
# string this app wrote; that is worth a dependency rather than hand-rolling.
# The sheet shown in the browser is SVG and does not need this - a missing
# ezdxf costs the DXF download and nothing else.
ezdxf
